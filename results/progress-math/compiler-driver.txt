import {spawnSync} from 'node:child_process';
import {mkdirSync,writeFileSync,statfsSync} from 'node:fs';
const root='O:/xterm-rust-swarm-16/tmp/worktrees/BradleyProgress551';
const output=root+'/results/progress-math';
const temporary=root+'/tmp/compiler';mkdirSync(temporary,{recursive:true});
const cargo='C:/Users/user/.cargo/bin/cargo.exe';
const env={...process.env,CARGO_BUILD_JOBS:'2',CARGO_TARGET_DIR:root+'/target',TEMP:temporary,TMP:temporary,TMPDIR:temporary};
const report={root,started:new Date().toISOString(),steps:[]};
const commands=[
 ['native-progress-status',['test','-p','xterm-addons','--all-features','--test','progress_upstream','--test','status_live_terminal','-j2']],
 ['native-web-lib',['test','-p','xterm-web','--lib','--all-features','-j2']],
 ['native-web-clippy',['clippy','-p','xterm-web','--lib','--all-features','-j2','--','-D','warnings']],
 ['wasm-clippy',['clippy','-p','xterm-web','--target','wasm32-unknown-unknown','--all-features','--lib','--test','wasm_progress_math','--test','wasm_status_coercion','--test','wasm_status_addons','--test','wasm_status_clear','-j2','--','-D','warnings']],
 ['wasm-no-run',['test','-p','xterm-web','--target','wasm32-unknown-unknown','--all-features','--test','wasm_progress_math','--test','wasm_status_coercion','--test','wasm_status_addons','--test','wasm_status_clear','-j2','--no-run']]
];
try {
 for(const [name,args] of commands){
  const probe=spawnSync('C:/Windows/System32/WindowsPowerShell/v1.0/powershell.exe',['-NoProfile','-Command','(Get-CimInstance Win32_OperatingSystem).FreePhysicalMemory*1024'],{encoding:'utf8',timeout:30000});
  const ram=Number(probe.stdout.trim()),disk=statfsSync('O:/',{bigint:true}),free=disk.bavail*disk.bsize;
  if(probe.status!==0||!Number.isFinite(ram)||ram<8*1024**3||free<10n*1024n**3n)throw Error('resource floor failed');
  const start=Date.now(),result=spawnSync(cargo,args,{cwd:root,env,encoding:'utf8',timeout:240000,maxBuffer:32*1024**2,windowsHide:true});
  writeFileSync(output+'/'+name+'.log',(result.stdout??'')+(result.stderr??''));
  const step={name,command:[cargo,...args],ramBytes:ram,diskFreeBytes:free.toString(),exit:result.status,error:result.error?.message,elapsedMs:Date.now()-start};
  report.steps.push(step);console.log(JSON.stringify(step));
  if(result.status!==0||result.error)throw Error(name+' failed');
 }
 report.status='passed';
}catch(error){report.status='failed';report.error=String(error.stack??error);process.exitCode=1;}
finally {report.finished=new Date().toISOString();writeFileSync(output+'/compiler.json',JSON.stringify(report,null,2)+'\n');}
