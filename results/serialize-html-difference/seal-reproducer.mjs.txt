import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFile, writeFile, readdir, lstat } from 'node:fs/promises';
import { join, resolve } from 'node:path';
import { sourceIdentity, verifyBuiltPackage } from '../scripts/e2e/build-identity.mjs';
import { referenceIntegrity } from '../scripts/upstream-baseline-utils.mjs';
const root = resolve(import.meta.dirname, '..'), relative = 'results/serialize-html-difference';
const dir = join(root, relative);
const pins = async paths => Promise.all(paths.sort().map(async path => {
  const bytes = await readFile(join(root, path));
  return { path, bytes: bytes.length, sha256: createHash('sha256').update(bytes).digest('hex') };
}));
async function files(path) {
  const list = [];
  for (const entry of await readdir(join(root, path), { withFileTypes: true })) {
    assert(!entry.isSymbolicLink()); const name = `${path}/${entry.name}`;
    if (entry.isDirectory()) list.push(...await files(name)); else { assert(entry.isFile()); list.push(name); }
  }
  return list;
}
const source = await sourceIdentity(root); const build = await verifyBuiltPackage(root, source);
const manifest = JSON.parse(await readFile(join(root, 'docs/baseline/reference-manifest.json')));
const reference = {};
for (const [label, path] of [['original', process.env.SOURCE_ROOT], ['physical', join(root, 'source_repo')]]) {
  reference[label] = referenceIntegrity(path, manifest); assert.equal(reference[label].mismatches.length, 0);
}
const stages = [];
for (const stage of ['before', 'initial-fixed']) {
  const report = JSON.parse(await readFile(join(dir, stage, 'verification.json')));
  const browsers = [];
  for (const browser of report.browsers) {
    const left = JSON.parse(await readFile(join(dir, stage, `${browser.name}-source.json`)));
    const right = JSON.parse(await readFile(join(dir, stage, `${browser.name}-rust.json`)));
    assert.equal(left.length, 96); assert.equal(right.length, 96);
    const mismatches = [], groups = [];
    for (let index = 0; index < left.length; index++) {
      if (JSON.stringify(left[index]) !== JSON.stringify(right[index])) mismatches.push(left[index].name);
      if (!left[index].name.startsWith('html-raw-row-subtraction-')) continue;
      const expected = Array.isArray(left[index].result) ? left[index].result : left[index].result.values;
      const actual = Array.isArray(right[index].result) ? right[index].result : right[index].result.values;
      groups.push({ name: left[index].name, executed: expected.length,
        passed: expected.filter((value, i) => JSON.stringify(value) === JSON.stringify(actual[i])).length });
    }
    assert.equal(mismatches.length, stage === 'before' ? 3 : browser.name === 'firefox' ? 1 : 0);
    browsers.push({ name: browser.name, version: browser.version, executed: 96, passed: 96 - mismatches.length,
      failed: mismatches.length, mismatches, groups });
  }
  stages.push({ stage, reportStatus: report.status, seed: report.seed, seededCases: report.seededCases,
    browsers, suites: report.suites.map(({ engine, counts, result }) => ({ engine, counts, result })) });
}
await writeFile(join(dir, 'summary.json'), JSON.stringify({ source, build, reference, stages }, null, 2) + '\n');
const runtime = await files('browser/pkg');
runtime.push('browser/xterm.css', 'target/wasm32-unknown-unknown/debug/xterm_web.wasm');
const executables = (await files('target/debug/deps')).filter(path => /\/xterm_web-[a-f0-9]+\.exe$/.test(path));
assert.equal(executables.length, 1); runtime.push(...executables, 'results/demos/build/latest.json');
runtime.push(...await files(build.report.slice(0, build.report.lastIndexOf('/'))));
await writeFile(join(dir, 'runtime-pins.json'), JSON.stringify(await pins(runtime), null, 2) + '\n');
const producer = [
  'crates/xterm-web/src/serialize_addon/html.rs', 'crates/xterm-web/src/serialize_addon/options.rs',
  'scripts/serialize-html-difference-cases.mjs', 'scripts/serialize-browser-check.mjs', 'scripts/serialize-browser-cases.mjs',
  'docs/serialize-html-difference.md', 'scripts/e2e/build-bootstrap.mjs', 'scripts/e2e/fixture-server.mjs', 'scripts/e2e/test-page.mjs',
  ...await files(relative)
].filter(path => !path.endsWith('/producer-pins.json'));
await writeFile(join(dir, 'producer-pins.json'), JSON.stringify(await pins(producer), null, 2) + '\n');
console.log(`Sealed source${source.files} and ${producer.length} producer/${runtime.length} runtime pins`);
