async function command(name, executable, args, expected = 0, extraEnv = {}) {
  const disk = await statfs(root); const ram = freemem();

    timeoutMs: 300000, signal: abort.signal, log: resolve(out, name + '.log'), resourceCheck: async () => {
      const disk = await statfs(root);
      if (freemem() < 8 * 1024 ** 3 || disk.bavail * disk.bsize < 10 * 1024 ** 3) throw Error('headroom dropped');
    } });
  const output = await readFile(resolve(out, name + '.log'), 'utf8');
  report.commands.push({ name, executable, args, ramBytes: ram, diskBytes: disk.bavail * disk.bsize,
    ...result, status: result.exitCode, durationMs: Date.now() - start,
    logSha256: hash(output), summaries: [...output.matchAll(/test result: (?:ok|FAILED)\. (\d+) passed; (\d+) failed; (\d+) ignored; (\d+) measured; (\d+) filtered out/g)].map(m => m.slice(1).map(Number)) });
  await writeFile(resolve(out, phase + '.json'), JSON.stringify(report, null, 2) + '\n');
  console.log(name, result.exitCode, result.error ?? '');
  if (result.error || result.signal || result.timedOut || result.resourceError || result.terminationError || result.exitCode !== expected) throw Error(name + ': unexpected outcome');
}
