// Reuses Andrea456's bounded source/package swap for a separate HTML repair.
import assert from 'node:assert/strict';
import {readFile,writeFile,mkdir,copyFile,cp,rm} from 'node:fs/promises';
import {dirname,join} from 'node:path';
import {sourceIdentity,verifyBuiltPackage} from '../scripts/e2e/build-identity.mjs';
import {runCommand} from '../scripts/quality-process.mjs';
const root=process.cwd();
const files=['crates/xterm-web/src/serialize_addon/html.rs','results/demos/build/latest.json'];
async function copy(file,target){await mkdir(dirname(target),{recursive:true});await copyFile(file,target);}
async function snapshot(name){for(const file of files)await copy(file,join('tmp',name,file));await cp('browser/pkg',join('tmp',name,'browser/pkg'),{recursive:true});}
async function restore(name){for(const file of files)await copy(join('tmp',name,file),file);await rm('browser/pkg',{recursive:true});await cp(join('tmp',name,'browser/pkg'),'browser/pkg',{recursive:true});}
async function capture(name,suite){
  const dest=join('results/serialize-oversized-column',name);await mkdir(dest,{recursive:true});
  const identity=await sourceIdentity(root);await verifyBuiltPackage(root,identity);
  const result=await runCommand(process.execPath,['scripts/serialize-browser-check.mjs',...(suite?['--suite']:[])],{cwd:root,env:process.env,log:join(dest,'runner.log'),timeoutMs:360000});
  await writeFile(join(dest,'runner-result.json'),JSON.stringify(result,null,2)+'\n');
  const outputs=['verification.json',...['chromium','firefox','webkit'].flatMap(browser=>[`${browser}-source.json`,`${browser}-rust.json`,`${browser}-terminal.png`])];
  if(suite)for(const engine of ['source','rust'])for(const file of ['playwright.json','playwright.log','configuration.cjs.txt'])outputs.push(`${engine}-suite/${file}`);
  for(const file of outputs)await copy(join('results/serialize-browser',file),join(dest,file));
  const report=JSON.parse(await readFile(join(dest,'verification.json')));
  console.log(name,JSON.stringify({identity,browsers:report.browsers.map(b=>({name:b.name,passed:b.passed,executed:b.executed,mismatches:b.mismatches.map(m=>m.name),error:b.error})),suites:report.suites.map(s=>({engine:s.engine,counts:s.counts}))}));
  return report;
}
const fixedIdentity=await sourceIdentity(root);await verifyBuiltPackage(root,fixedIdentity);await snapshot('fixed');
try{
  await restore('baseline');const baseline=await capture('baseline',false);
  assert.equal(baseline.status,'failed');assert.equal(baseline.browsers.length,3);
  for(const b of baseline.browsers){assert.equal(b.executed,68);assert.deepEqual(b.mismatches.map(m=>m.name).sort(),['html-oversized-start-column-header-throw','html-oversized-start-column-reentry','raw-numeric-boundaries']);}
}finally{await restore('fixed');assert.equal((await sourceIdentity(root)).sha256,fixedIdentity.sha256);await verifyBuiltPackage(root,fixedIdentity);}
const fixed=await capture('fixed',true);
assert.equal(fixed.status,'failed');assert.equal(fixed.browsers.length,3);
for(const b of fixed.browsers){assert.equal(b.executed,68);assert.deepEqual(b.mismatches.map(m=>m.name),['raw-numeric-boundaries']);}
assert.equal(fixed.suites.length,2);for(const s of fixed.suites){assert.equal(s.counts.passed,120);assert.equal(s.counts.failed,0);}
assert.equal((await sourceIdentity(root)).sha256,fixedIdentity.sha256);await verifyBuiltPackage(root,fixedIdentity);
