import assert from 'node:assert/strict';
import { lstat, realpath, readdir, readFile, writeFile } from 'node:fs/promises';
import { execFileSync } from 'node:child_process';
import { resolve, join } from 'node:path';
import { referenceIntegrity } from '../scripts/upstream-baseline-utils.mjs';
const root = process.cwd(), links = [], redirects = [];
let files = 0, directories = 0;
async function scan(path) {
  const stat = await lstat(path);
  if (stat.isSymbolicLink()) { links.push(path); return; }
  if (stat.isDirectory()) {
    directories++;
    if (resolve(await realpath(path)).toLowerCase() !== resolve(path).toLowerCase()) { redirects.push(path); return; }
    for (const entry of await readdir(path)) await scan(join(path, entry));
  } else if (stat.isFile()) files++;
}
await scan(root);
const powershell = 'C:/Windows/System32/WindowsPowerShell/v1.0/powershell.exe';
const escaped = root.replaceAll("'", "''");
const command = `$root='${escaped}'; @(Get-CimInstance Win32_Process | Where-Object { $_.ProcessId -ne ${process.pid} -and $_.Name -match '^(cargo|rustc|clippy-driver|node|chrome|firefox|MiniBrowser|WebKitNetworkProcess|WebKitWebProcess)(\\.exe)?$' -and $_.CommandLine -and ($_.CommandLine.Replace('\\','/').ToLower().Contains($root.Replace('\\','/').ToLower())) } | Select-Object ProcessId,Name,CommandLine) | ConvertTo-Json -Compress`;
const text = execFileSync(powershell, ['-NoProfile','-Command', command], { encoding:'utf8', timeout:30000 }).trim();
const processes = text ? JSON.parse(text) : [];
const list = Array.isArray(processes) ? processes : [processes];
const manifest = JSON.parse(await readFile('docs/baseline/reference-manifest.json'));
const reference = referenceIntegrity(process.env.SOURCE_ROOT, manifest);
assert.equal(links.length, 0); assert.equal(redirects.length, 0); assert.equal(list.length, 0); assert.equal(reference.mismatches.length, 0);
const result = { root, files, directories, links, redirects, processes:list, reference, sourceBranch:'swarm/AndreaAnsi', reviewedTip:'c0f6dfe7', ownedTemporaryRoot:root, heavyGrantsHeld:0 };
await writeFile('results/serialize-ansi-numeric/cleanup-preflight.json', JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result));
