async function stage(name,browsers,expected){
  console.log('START '+name);
  const record=await runCommand(process.execPath,['scripts/browser-host-check.mjs'],{cwd:root,env:{...env,BROWSERS:browsers},log:resolve(out,name+'.log'),timeoutMs:600000});
  const latest=JSON.parse(await readFile(resolve(root,'results/browser-host/render_service_browser-results.json'),'utf8'));
  const folder=resolve(out,name);await mkdir(folder,{recursive:true});
  if (!latest.artifacts) throw Error('browser build has no artifacts: ' + latest.error);
  const run=resolve(latest.artifacts.wasm.path,'..','..');
  // Only raw reports/logs are permanent. Generated artifacts stay available in
  // the owned worktree through review and are pinned, then removed on cleanup.
  for(const entry of await readdir(run,{withFileTypes:true}))if(entry.isFile())await copyFile(join(run,entry.name),join(folder,entry.name));
  const pins=[];
  async function pinTree(dir){for(const entry of await readdir(dir,{withFileTypes:true})){const p=join(dir,entry.name);if(entry.isDirectory())await pinTree(p);else if(entry.isFile())pins.push({path:p,sha256:hash(await readFile(p))});else throw Error('unexpected generated link '+p);}}
  if(latest.artifacts){await pinTree(resolve(run,'artifacts'));pins.push({path:latest.artifacts.compiler.originalPath,sha256:latest.artifacts.compiler.sha256});}
  const result={name,productionSha256:hash(await readFile(file)),record,report:latest,pins};records.push(result);
  await writeFile(resolve(out,'browser-stages.json'),JSON.stringify(records,null,2)+'\n');
  console.log(name+' '+record.status+' '+JSON.stringify(latest.browsers.map(b=>({name:b.name,status:b.status,counts:b.counts}))));
  if(latest.status!==expected)throw Error('unexpected status '+name);
  if(expected==='passed'&&record.status!=='passed')throw Error('runner failure '+name);
  if(expected==='failed'&&!latest.browsers.some(b=>b.counts?.failed>0))throw Error('negative control lacked assertion failure '+name);
}
